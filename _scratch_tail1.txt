const glBucketRowEl = document.getElementById('glBucketRow');

function renderGlBucketMetrics(){
  glBucketRowEl.innerHTML = '';
  const matched = GL_BUCKETS
    .map(bucket => ({ bucket, gls: bucket.gls.filter(gl => activeGL.has(gl)) }))
    .filter(m => m.gls.length);
  if (!matched.length){
    glBucketRowEl.style.display = 'none';
    return;
  }
  glBucketRowEl.style.display = 'flex';
  matched.forEach(({bucket, gls}) => {
    const count = DATA.filter(r => gls.includes(r.gl_name)).length;
    const card = document.createElement('div');
    card.className = 'glbucket-card';

    const label = document.createElement('div');
    label.className = 'glbucket-label';
    label.textContent = bucket.label + ' — ' + gls.join(', ');
    card.appendChild(label);

    const countEl = document.createElement('div');
    countEl.className = 'glbucket-count';
    countEl.textContent = count;
    const suffix = document.createElement('span');
    suffix.textContent = count === 1 ? 'seller' : 'sellers';
    countEl.appendChild(suffix);
    card.appendChild(countEl);

    const metricsEl = document.createElement('div');
    metricsEl.className = 'glbucket-metrics';
    bucket.metrics.forEach(m => {
      const row = document.createElement('div');
      row.className = 'glbucket-metric-row';
      const k = document.createElement('span');
      k.textContent = m;
      const v = document.createElement('span');
      v.className = 'v';
      v.textContent = 'TBD';
      row.appendChild(k);
      row.appendChild(v);
      metricsEl.appendChild(row);
    });
    card.appendChild(metricsEl);

    glBucketRowEl.appendChild(card);
  });
}

function buildGLChips(){
  glChipsEl.innerHTML = '';
  glNames.forEach(name => {
    const btn = document.createElement('button');
    btn.type = 'button';
    btn.className = 'gl-chip';
    btn.textContent = name;
    btn.setAttribute('aria-pressed', String(activeGL.has(name)));
    btn.addEventListener('click', () => {
      if (activeGL.has(name)) activeGL.delete(name); else activeGL.add(name);
      btn.setAttribute('aria-pressed', String(activeGL.has(name)));
      closePopover();
      buildHead(); // gl_name/gm_name columns appear or disappear with the toggle
      renderGlBucketMetrics();
      render();
    });
    glChipsEl.appendChild(btn);
  });
}

function valueToFilterString(v){
  return (v === null || v === undefined || v === '') ? '__BLANK__' : String(v);
}

function distinctValuesFor(key){
  const vals = new Set(DATA.map(r => valueToFilterString(r[key])));
  const arr = [...vals];
  arr.sort((a, b) => {
    if (a === '__BLANK__') return 1;
    if (b === '__BLANK__') return -1;
    return a.localeCompare(b, undefined, {numeric: true, sensitivity: 'base'});
  });
  return arr;
}

let popoverKey = null;
let currentFilterBtn = null;

function updateFilterBtnState(){
  if (currentFilterBtn) currentFilterBtn.classList.toggle('active', !!colFilterValues[popoverKey]);
}

function applyPopoverSearch(){
  const q = popoverSearchEl.value.trim().toLowerCase();
  popoverListEl.querySelectorAll('.popover-item').forEach(item => {
    const text = item.querySelector('span').textContent.toLowerCase();
    item.style.display = text.includes(q) ? '' : 'none';
  });
}

function openPopoverFor(key, btn){
  popoverKey = key;
  currentFilterBtn = btn;
  const all = distinctValuesFor(key);
  const selected = colFilterValues[key] ? colFilterValues[key] : new Set(all);
  popoverSearchEl.value = '';
  popoverListEl.innerHTML = '';
  all.forEach(v => {
    const item = document.createElement('label');
    item.className = 'popover-item';
    const cb = document.createElement('input');
    cb.type = 'checkbox';
    cb.checked = selected.has(v);
    cb.addEventListener('change', () => {
      if (!colFilterValues[key]) colFilterValues[key] = new Set(all);
      if (cb.checked) colFilterValues[key].add(v); else colFilterValues[key].delete(v);
      if (colFilterValues[key].size === all.length) delete colFilterValues[key];
      updateFilterBtnState();
      render();
    });
    const span = document.createElement('span');
    span.textContent = v === '__BLANK__' ? '(Blanks)' : v;
