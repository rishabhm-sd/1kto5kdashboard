    });
  });
  if (sortKey){
    rows = rows.slice().sort((a,b) => {
      const av = a[sortKey], bv = b[sortKey];
      const an = Number(av), bn = Number(bv);
      let cmp;
      if (!Number.isNaN(an) && !Number.isNaN(bn) && av !== '' && bv !== ''){
        cmp = an - bn;
      } else {
        cmp = String(av ?? '').localeCompare(String(bv ?? ''));
      }
      return cmp * sortDir;
    });
  }
  const cols = visibleColumns();
  bodyRows.innerHTML = '';
  rows.forEach(r => {
    const tr = document.createElement('tr');
    cols.forEach(c => {
      const td = document.createElement('td');
      td.textContent = r[c.key] ?? '';
      tr.appendChild(td);
    });
    tr.addEventListener('click', () => openSellerModal(r));
    bodyRows.appendChild(tr);
  });
  rowcountEl.textContent = rows.length + (rows.length === 1 ? ' seller' : ' sellers') +
    (rows.length === DATA.length ? '' : ' of ' + DATA.length) +
    (DATA.length ? '' : ' (no data loaded)');
  tableEl.style.display = DATA.length ? 'table' : 'none';
  emptyState.style.display = DATA.length ? 'none' : 'block';
  lastRenderedRows = rows;
}

// ---- CSV export (whatever is currently on screen: same filters/sort/GL
// toggle and visible columns as the table, so the download always matches
// what the user is looking at). ----
function toCsvValue(v){
  if (v === null || v === undefined) return '';
  const s = String(v);
  if (/[",\n]/.test(s)) return '"' + s.replace(/"/g, '""') + '"';
  return s;
}
function downloadCsv(){
  const cols = visibleColumns();
  const lines = [cols.map(c => toCsvValue(c.label)).join(',')];
  lastRenderedRows.forEach(r => {
    lines.push(cols.map(c => toCsvValue(r[c.key])).join(','));
  });
  const blob = new Blob([lines.join('\r\n')], {type: 'text/csv;charset=utf-8;'});
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = 'all-sellers-' + new Date().toISOString().slice(0,10) + '.csv';
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
}
downloadCsvBtn.addEventListener('click', downloadCsv);
